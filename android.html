Quick start

// Your Activity / Composable
val (accent, accentInk) = HardbrutTokens.AccentYellow

HardbrutButton(
    onClick = { /* do something */ },
    accent = accent,
    accentInk = accentInk
) { Text("CLICK ME") }

HardbrutButton(
    onClick = { /* dismiss */ },
    accent = accent,
    accentInk = accentInk,
    kind = ButtonKind.Cancel
) { Text("CANCEL") }

HardbrutCard {
    Text("Hello from HARDBRUT")
}

Hard shadow helper

fun Modifier.hardShadow(size: Dp, color: Color = Color.Black): Modifier = this.then(
    drawBehind {
        drawRect(color = color, topLeft = Offset(size.toPx(), size.toPx()),
            size = Size(this.size.width, this.size.height))
    }
)
